15const DNS_OVERRIDE:&str = include_str!("../Resource/dns-override.js");

17/// Spawns a Node.js sidecar with DNS override configured to use the local
18/// Hickory DNS server.
19///
20/// This function:
21/// 1. Creates the app data directory if it doesn't exist
22/// 2. Writes the DNS override JavaScript file to the app data directory
23/// 3. Configures the sidecar process with NODE_OPTIONS to require the DNS
24///    override script
25/// 4. Sets the Resolve environment variable with the local DNS server address
26/// 5. Spawns the sidecar process
27///
28/// # Parameters
29///
30/// * `app` - The Tauri app handle, used to access the app data directory and
31///   shell
32/// * `sidecar_name` - The name of the sidecar executable to spawn
33///
34/// # Returns
35///
36/// Returns `Ok(())` if the sidecar was spawned successfully, or an error if:
37/// - The app data directory couldn't be created or accessed
38/// - The DNS override file couldn't be written
39/// - The sidecar couldn't be found or spawned
40///
41/// # Example
42///
43/// ```rust,no_run
44/// use tauri::Manager;
45/// use SideCar::Spawn::spawn_node_sidecar;
46///
47/// #[tauri::command]
48/// fn launch_sidecar(app:tauri::AppHandle) -> Result<(), String> {
49/// 	spawn_node_sidecar(&app, "my-sidecar").map_err(|e| e.to_string())?;
50/// 	Ok(())
51/// }
52/// ```
53#[allow(dead_code)]
54pub fn spawn_node_sidecar(app:&AppHandle, sidecar_name:&str) -> anyhow::Result<()> {
55	// Ensure app data directory exists
56	let data_dir = app.path().app_data_dir()?;
57	fs::create_dir_all(&data_dir)?;
58
59	// Write DNS override script to app data directory
60	let override_path = data_dir.join("dns-override.js");
61	fs::write(&override_path, DNS_OVERRIDE)?;
62
63	// Get the DNS server port from Mist module
64	let port = dns_port();
65	let node_opts = format!("--require {}", override_path.display());
66
67	// Spawn the sidecar with DNS configuration
68	app.shell()
69		.sidecar(sidecar_name)?
70		.env("NODE_OPTIONS", &node_opts)
71		.env("Resolve", format!("127.0.0.1:{port}"))
72		.spawn()?;
73
74	Ok(())
75}
